        const fields = document.querySelectorAll('.common-field');
        const errorMessages = document.querySelectorAll('.common-error-message');
        const phoneField = document.querySelector('input[name="phone"]');
        const emailField = document.querySelector('input[name="email"]');

        const validateEmail = (email) => {
            const emailPattern = /^[^\s@]+@[^\s@]+\.[^\s@]+$/;
            return emailPattern.test(email);
        };
        const validatePhone = (phone) => {
            const phoneRegExp = /^\d{10}$/;
            const cleaned = phone.replace(/\D/g, '');
            return cleaned[0] !== '1' && phoneRegExp.test(cleaned);
        };
        const formatPhoneNumber = (phone) => {
            phone = phone.replace(/\D/g, "");
            if (phone.length === 10) {
                return phone.replace(/^(\d{3})(\d{3})(\d{4})$/, "($1) $2-$3");
            } else if (phone.length > 6) {
                return phone.replace(/^(\d{3})(\d{3})(\d*)$/, "($1) $2-$3");
            } else if (phone.length > 3) {
                return phone.replace(/^(\d{3})(\d*)$/, "($1) $2");
            } else if (phone.length > 0) {
                return "(" + phone;
            }
            return phone;
        };

        emailField.addEventListener('input', function (e) {
            const emailInput = e.target;
            const email = emailInput.value;
            const errorMessage = document.getElementById('email-error-message');

            if (!validateEmail(email)) {
                errorMessage.textContent = 'Invalid email address';
                errorMessage.style.visibility = 'visible';
                emailInput.setCustomValidity('Invalid email address');
            } else {
                errorMessage.textContent = '-';
                errorMessage.style.visibility = 'hidden';
                emailInput.setCustomValidity('');
            }
        });

            const onlyNumbers = e.target.value.replace(/\D/g, "");

            const phone = phoneInput.value;
            const errorMessage = document.getElementById('phone-error-message');

    function submitForm(formDataObject) {
        if (formDataObject.country === "Canada") {
            formDataObject.zip = formDataObject.canadaZip;
            formDataObject.state = formDataObject.canadaState;
        } else if (formDataObject.country === "United Kingdom") {
            formDataObject.zip = formDataObject.ukZip;
        } else if (formDataObject.country == "India") {
            if (formDataObject.levelOfEducation.includes("Bachelor's Degree")) {
                formDataObject.levelOfEducation = "Bachelor's Degree";
            } else if (formDataObject.levelOfEducation.includes("Master's Degree")) {
                formDataObject.levelOfEducation = "Master's Degree";
            }
        }

        if (formDataObject.country === "United States") {
            formDataObject.workAuthorization = formDataObject.workAuthorization === "yes" &&
                formDataObject.sponsorship === "no" &&
                formDataObject.futureSponsorship === "no" &&
                formDataObject.futureFamilySponsorship === "no"
                ? "Yes"
                : "No"
        } else if (["United Kingdom", "Canada"].includes(formDataObject.country)) {
            formDataObject.workAuthorization = formDataObject.workAuthorization === "yes" && formDataObject.caSponsorship === "no" ? "Yes" : "No";
        }
        const queryParams = getQueryParams();
        const standardizedQuery = standardizeQueryParams(queryParams);
        // Make POST request to the API endpoint
        const apiObject = {
            url: window?.location?.href.split('#')[0] || "",
            ApplicationDevice__c: window.innerWidth < 640 ? "Mobile" : "Desktop",
            irClickId: standardizedQuery?.irclickid || "",
            searchEngine: standardizedQuery?.searchengine || "",
            searchString: standardizedQuery?.srstring || "",
            payPerClickKeyword: standardizedQuery?.keyword || "",
            gCLID: standardizedQuery?.gclid || "",
            uTMTerm: standardizedQuery?.utm_term || "",
            uTMCampaign: standardizedQuery.utm_campaign || "",
            uTMContent: standardizedQuery.utm_content || "",
            uTMMedium: standardizedQuery.utm_medium || "",
            uTMSource: standardizedQuery.utm_source || "",
            uTMSchoolID: standardizedQuery.utm_schoolid || "",
            referrerURL: document.referrer || "Direct",
            uTMReferrerName: standardizedQuery.utm_referrername || "",
            campaignvalue: standardizedQuery.campaignvalue || "",
            appcastClickID: '',
            sourcedBy: standardizedQuery.sourcedby || "",
            referredByEmail: standardizedQuery.referredByEmail || "",
            referredBy : standardizedQuery.ra || "",
            referredByUser: standardizedQuery.ru || "",
            dropbox: '',
            veteran: (window?.location?.href.split('#')[0] || "").includes("veteran"),
            Resumedropbox: '',
            ...formDataObject,
            leadDate: new Date().toISOString(),
            phone: formDataObject.phone.replace(/\D/g, ""),
            FileBase64: '',
            FileExt: '',
            ResumeUpload: '',
            computer_data: '',
            computer_data_result: '',
            graduationDate: formDataObject.graduationMonth ? `${formDataObject.graduationYear}-${formDataObject.graduationMonth}-01` : "",
            dataConsent: formDataObject.dataConsent == "on",
            leadType: "Revature"
        }
        delete apiObject["computer_data"];
        delete apiObject["computer_data_result"];
        delete apiObject["g-recaptcha-response"];

        const apiUrl = ENV_VAR.B2C_FORM_API_ENDPOINT;
        const queryString = Object.keys(apiObject)
            .map(key => encodeURIComponent(key) + '=' + encodeURIComponent(apiObject[key]))
            .join('&');
        const apiUrlWithParams = apiUrl + '?' + queryString;

        fetch(apiUrlWithParams, {
            method: 'GET',
            mode: 'no-cors'
        }).then(response => {
            if (response.ok) {
                console.log('Form data submitted successfully');
            } else {
                console.error('Error submitting form data');
            }
            window.location.href = `/thank-you-for-submission?name=${btoa(apiObject?.firstName)}`;
        }).catch(error => {
            console.error('Error submitting form data:', error);
            alert('Error submitting form data, Please try again.');
            showSubmitButton()
        });
    }

    function handleDropboxFileChange(file) {
        const fileError = document.getElementById('fileError');
        const fileSuccess = document.getElementById('fileSuccess');
        fileError.textContent = '';
        fileSuccess.textContent = '';
        if (file.bytes > 5242880) {
            fileError.textContent = "File size is too large.";
        } else {
            document.getElementById('computer_data').value = null;
            document.getElementById('dropbox_auth_name').value = file.link;
            document.getElementById('Resumedropbox').value = file.link;
            document.getElementById('fileError').textContent = '';
            fileSuccess.textContent = 'Resume ready to upload';
        }
    }

    function handleFileChange(e) {
        let reader = new FileReader();
        let file = e.target.files[0];

        if (!file) {
            return;
        }
        fileError.textContent = '';
        reader.onloadend = function (e) {
            const result = e.target.result;
            const rawData = result.split("base64,")[1];
            const allowedExtensions = ["pdf", "doc", "docx", "rtf", "txt"];
            const fileExtension = file.name.split(".").pop();
            const fileError = document.getElementById('fileError');
            const fileSuccess = document.getElementById('fileSuccess');
            fileError.textContent = '';
            fileSuccess.textContent = '';
            if (!allowedExtensions.includes(fileExtension)) {
                fileError.textContent = "Invalid file type.";
            } else if (file.size > 5242880) {
                fileError.textContent = "File size is too large.";
            } else {
                document.getElementById('computer_data').value = file.name;
                document.getElementById('computer_data_result').value = rawData;
                document.getElementById('FileBase64').value = rawData;
                document.getElementById('FileExt').value = file.type;
                document.getElementById('dropbox_auth_name').value = null;
                fileSuccess.textContent = 'Resume ready to upload';
            }
        };

        reader.readAsDataURL(file);
    };

    function uploadResume() {
        // Collect form data
        const formData = new FormData(document.getElementById('myForm'));
        const formDataObject = {};
        formData.forEach((value, key) => {
            formDataObject[key] = typeof value != 'object' && !Array.isArray(value) && value !== null ? value?.trim() : value;
        });
        if (formDataObject.computer_data) {
            fetch(ENV_VAR.RESUME_API_ENDPOINT,
                {
                    method: "POST",
                    body: JSON.stringify({
                        key: "245583662863Rk863369",
                        person: `${formDataObject.firstName} ${formDataObject.lastName}`,
                        filename: formDataObject.computer_data,
                        file: formDataObject.computer_data_result,
                    }),
                },
            ).then((response) => response.json()).then((data) => {
                if (data.success) {
                    document.getElementById('resumeURL').value = data.link;
                    formDataObject.resumeURL = data.link;
                    submitForm(formDataObject);
                } else {
                    document.getElementById('fileError').textContent = 'There was an error uploading your file. Please try again.';
                    showSubmitButton()
                }
            }).catch((error) => {
                document.getElementById('fileError').textContent = 'There was an error uploading your file. Please try again.';
                showSubmitButton()
            });
        } else if (formDataObject.dropbox) {
            const extension = formDataObject.dropbox.split("/").pop().split("#")[0].split("?")[0];
            let url = formDataObject.dropbox.replace("dl=0", "dl=1");
            url = url?.trim();
            const base64 = btoa(url);
            fetch(ENV_VAR.RESUME_API_ENDPOINT,
                {
                    method: "POST",
                    body: JSON.stringify({
                        key: "245583662863Rk863369",
                        person: `${formDataObject.firstName} ${formDataObject.lastName}`,
                        filename: extension,
                        file: base64,
                    }),
                }).then((response) => response.json()).then((data) => {
                    if (data.success) {
                        document.getElementById('resumeURL').value = data.link;
                        formDataObject.resumeURL = data.link;
                        submitForm(formDataObject);
                    } else {
                        document.getElementById('fileError').textContent = 'There was an error uploading your file. Please try again.';
                        showSubmitButton()
                    }
                })
                .catch((error) => {
                    document.getElementById('fileError').textContent = 'There was an error uploading your file. Please try again.';
                    showSubmitButton()
                });
        } else {
            document.getElementById('fileError').textContent = "Please upload a resume.";
            showSubmitButton()
        }
    }

    function showSubmitButton() {
        document.getElementById('b2b-form-button').style.display = 'block';
        document.getElementById('loadSpinner').classList.remove('spinner');
    }

    function radioButtonsValidation(radioButtons, errorMessage) {
        if (radioButtons && errorMessage) {
            let isSelected = false;
            radioButtons.forEach(radio => {
                if (radio.checked) {
                    isSelected = true;
                }
            });

            if (!isSelected) {
                errorMessage.textContent = "Please select an option.";
                errorMessage.style.visibility = 'visible';
            } else {
                errorMessage.textContent = "-";
                errorMessage.style.visibility = 'hidden';
            }
        }
    }

    function setAutoComplete(input, list, Id_input, items) {
        items.sort((a, b) => {
            if (a.label < b.label) {
                return -1;
            }
            if (a.label > b.label) {
                return 1;
            }
            return 0;
        });
        if (input && list) {
            const updateAutocomplete = () => {
                const value = input.value.toLowerCase();
                list.innerHTML = ''; // Clear the list

                if (true) {
                    const filteredItems = items
                        .filter(item => item.label.toLowerCase().includes(value))
                        .sort((a, b) => {
                            const aLower = a.label.toLowerCase();
                            const bLower = b.label.toLowerCase();
                            return aLower.indexOf(value) - bLower.indexOf(value);
                        });

                    filteredItems.forEach(item => {
                        const div = document.createElement('div');
                        div.className = 'autocomplete-item';
                        div.textContent = item.label;
                        div.addEventListener('click', () => {
                            input.value = item.label;
                            Id_input ? Id_input.value = item.value : '';
                            list.innerHTML = ''; // Clear the list
                        });
                        list.appendChild(div);
                    });
                }
            }

            input.addEventListener('input', updateAutocomplete);
            input.addEventListener('focus', updateAutocomplete);

            document.addEventListener('click', (e) => {
                if (!list.contains(e.target) && e.target !== input) {
                    list.innerHTML = ''; // Close the list
                }
            });
        }
    }

    function getQueryParams() {
        const queryParamsString = window.location.search;
        const queryParams = new URLSearchParams(queryParamsString);
        return queryParams;
    }

    function standardizeQueryParams(queryParams) {
        const standardizedQueryParams = {};

        queryParams.forEach((value, key) => {
            if (key === 'slug') {
                return;
            }
            const lowercasedKey = key.toLowerCase();
            standardizedQueryParams[lowercasedKey] = lowercasedKey.includes('utm') ? value.toLowerCase() : value;
        });

        return standardizedQueryParams;
    }
